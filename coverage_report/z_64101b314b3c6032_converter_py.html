69 def _apply_global_attributes(self, text: str) -> str: 

70 """ 

71 Replaces raw product names with Antora attributes. 

72 Uses negative lookbehind/lookahead to protect URLs and file paths. 

73 

74 Args: 

75 text (str): The input string to process. 

76 

77 Returns: 

78 str: The text with product names replaced by attributes. 

79 """ 

80 # Guard clause: If text is empty or if attribute replacement is disabled, return the original text. 

81 if not text or not self.REPLACE_WITH_ATTRIBUTES: 

82 return text 

83 

84 # We iterate through the attribute map and apply replacements.  

85 # The regex ensures we only replace standalone occurrences of the product names,  

86 # not when they are part of URLs or file paths.  

87 for raw_name, attr in self.attribute_map.items(): 

88 # SMART REGEX: 

89 # (?<![/:]) -> Negative Lookbehind: Don't match if preceded by / or : (Shields paths/URLs) 

90 # \b...\b -> Word Boundary: Match the whole word only 

91 # (?![/.]) -> Negative Lookahead: Don't match if followed by / or . (Shields extensions/paths) 

92 pattern = rf"(?<![/:])\b{re.escape(raw_name)}\b(?![/.])" 

93 

94 text = re.sub(pattern, attr, text) 

95 

96 return text

99 def _slugify(self, text: str) -> str: 

100 """ 

101 Converts a heading title into a SEO-friendly, unique ID. 

102 

103 Args: 

104 text (str): The heading text to slugify. 

105 Returns: 

106 str: A slugified version of the heading, suitable for use as an ID. 

107 """ 

108 # Guard clause for empty text to prevent generating empty IDs 

109 lookup_key = text.strip() 

110 

111 if "VLT" in lookup_key and "HVAULT" in lookup_key: 

112 if hasattr(self, 'heading_vault') and lookup_key in self.heading_vault: 

113 text = self.heading_vault[lookup_key] 

114 

115 # 1. Strip raw HTML tags (if any survived) 

116 clean_text = re.sub(r'<[^>]+>', '', text) 

117 

118 # 2. Strip AsciiDoc roles (e.g., [.class-name]#Text#) 

119 clean_text = re.sub(r'\[\.[^\]]+\]#.*?#', '', clean_text) 

120 

121 # 3. Normalize to lowercase and replace spaces with hyphens 

122 slug = clean_text.lower() 

123 

124 # 4. Remove any remaining curly brace IDs and replace dots with hyphens 

125 slug = re.sub(r'\{#.*?\}', '', slug) 

126 slug = slug.replace('.', '-') 

127 slug = re.sub(r'[^a-z0-9-]', '-', slug) 

128 slug = re.sub(r'-+', '-', slug).strip('-') 

129 

130 # 5. Ensure uniqueness by appending a counter if needed 

131 base_slug = slug or "section" 

132 final_slug = base_slug 

133 

134 counter = 1 

135 

136 # 6. If the slug already exists, append a counter until we find a unique one 

137 while final_slug in self.used_ids: 

138 final_slug = f"{base_slug}-{counter}" 

139 counter += 1 

140 

141 # 7. Register the final slug to prevent future collisions 

142 self.used_ids.add(final_slug) 

143 

144 return final_slug

148 """ 

149 Prepares Markdown for Pandoc by shielding modern syntax and extracting metadata. 

150 Includes a 'Vault' mechanism to protect version headers from line-splitting. 

151 

152 Args: 

153 content (str): Raw Markdown string. 

154 

155 Returns: 

156 str: "Shielded" Markdown ready for Pandoc. 

157 """ 

158 self.metadata = {} 

159 self.discovered_title = None 

160 self.used_ids = set() 

161 self.protected_json = [] 

162 self.heading_vault = {}

164 def vault_header(m: re.Match) -> str: 

165 """ 

166 Protects version headers like "## 2.2.0 - 2026-03-23" from being split by the list item regex. 

167 

168 Args: 

169 m: The regex match object for a version header. 

170 """ 

171 # We store the raw header text in a vault and replace it with a unique key. 

172 level = m.group(1) 

173 

174 # We want to preserve the exact text of the header, including the version and date, so we store it in the vault. 

175 raw_text = m.group(2).strip() 

176 

177 # We generate a unique key for this header and store the raw text in the vault under that key. 

178 key = f"VLT{len(self.heading_vault)}HVAULT" 

179 

180 # We store the raw header text in the vault using the unique key. 

181 self.heading_vault[key] = raw_text 

182 

183 return f"{level} {key}" 

184 

185 # Secure the version headers before anything else touches them 

186 # We look for headers that start with ## (or more) followed by a version pattern and a date, and we vault them. 

187 pattern = r'^(#+)\s+(\d+\.[\d\.]+\d+\s+-\s+\d{4}-\d{2}-\d{2}).*$' 

188 # We replace the matched header with a vault key, and store the original header text  

189 # in the heading_vault for later restoration. 

190 content = re.sub(pattern, vault_header, content, flags=re.M) 

191 

192 # --- EXISTING SHIELDING LOGIC --- 

193 # 1. Code Blocks: Shield fenced code blocks to prevent Pandoc from mangling them. 

194 content = re.sub(r'(`{3}.*?`{3})', lambda m: m.group(1).replace('#', 'HASHSHIELD'), content, flags=re.DOTALL) 

195 # 2. Inline Code: Shield inline code to prevent accidental ID processing. 

196 content = re.sub(r'(`.*?`)', lambda m: m.group(1).replace('#', 'HASHSHIELD'), content) 

197 

198 # 3. Iframes: Temporarily replace iframes with a token to prevent Pandoc from stripping them. 

199 frontmatter_match = re.match(r'^---\s*\n(.*?)\n---\s*\n', content, re.DOTALL) 

200 

201 # 4. Custom Shielding Patterns: Apply any user-defined regex patterns for shielding. 

202 if frontmatter_match: 

203 try: 

204 import yaml 

205 

206 # We attempt to parse the frontmatter as YAML to extract metadata.  

207 # If successful, we store it in self.metadata. 

208 # If the frontmatter is not valid YAML, we catch the exception and simply leave  

209 # self.metadata as an empty dictionary, allowing the conversion to proceed without metadata. 

210 yaml_data = yaml.safe_load(frontmatter_match.group(1)) 

211 

212 # We check if the parsed YAML data is a dictionary before assigning it to self.metadata. 

213 if isinstance(yaml_data, dict): 

214 self.metadata = yaml_data 

215 

216 # If a title is found in the metadata, we can remove it from the content to prevent duplication. 

217 content = content[frontmatter_match.end():] 

218 except Exception: 

219 # If YAML parsing fails, we simply ignore the frontmatter and proceed with an empty metadata dictionary. 

220 self.metadata = {} 

221 

222 # 5. Title Discovery: If no title is in metadata, infer from the first H1 and remove it from content. 

223 self.discovered_title = self.metadata.get('title') 

224 

225 # We look for the first H1 header in the content to use as the title if it wasn't provided in the metadata. 

226 if not self.discovered_title: 

227 # We search for a line that starts with a single '#' followed by the title text. 

228 h1_match = re.search(r'^#\s+(.*)$', content, re.M) 

229 

230 # If we find an H1 header, we set it as the discovered title and remove that line  

231 # from the content to prevent duplication in the final AsciiDoc. 

232 if h1_match: 

233 self.discovered_title = h1_match.group(1).strip() 

234 # We remove the first occurrence of this H1 header from the content. 

235 content = content.replace(h1_match.group(0), "", 1) 

236 

237 # --- 6. HTML SHIELDING --- 

238 self.protected_html_blocks = []

240 placeholder = f"HTMLBLOCKSHIELDA{len(self.protected_html_blocks)}A" 

241 self.protected_html_blocks.append(match.group(1))

244 # Shield entire <style> or <script> blocks so Pandoc ignores their contents 

245 content = re.sub(r'(<style.*?</style>)', shield_html_block, content, flags=re.DOTALL|re.IGNORECASE)

250 # 7. Shielding Structural HTML Tags: We define a function to shield structural HTML tags like <div>, <ul>, and <li>. 

251 def shield_layout_tag(match: re.Match) -> str: 

252 """ 

253 Shields structural HTML tags like <div>, <ul>, and <li> to prevent Pandoc from stripping them. 

254 Uses a vault mechanism similar to the version headers to store the original tag content and  

255 replace it with a unique placeholder. This allows us to restore the exact original HTML tags after Pandoc processing, 

256 without risking any regex collisions or unintended replacements. 

257 

258 The placeholder format is designed to be unique and easily identifiable for restoration in the post-processing phase  

259 without interfering with any other content. By using a vault, we can ensure that even if the same HTML tag appears multiple times,  

260 each instance will be stored and restored correctly without any risk of collision or misreplacement.  

261 

262 Args: 

263 match: The regex match object for an HTML tag to be shielded. 

264 

265 Returns: 

266 str: A unique placeholder string that will be used in the content to represent the original HTML tag.  

267 """ 

268 placeholder = f"HTMLTAGSHIELDA{len(self.protected_html_tags)}A" 

269 self.protected_html_tags.append(match.group(1)) 

270 return placeholder 

271 

272 # 8a. Shield structural tags (div, ul, li) but let Pandoc handle <span> natively 

273 content = re.sub(r'(</?(?:div|ul|li)[^>]*>)', shield_layout_tag, content, flags=re.IGNORECASE) 

274 

275 # 8b. Shield <kbd> tags safely without applying HTML passthroughs (+++) 

276 content = re.sub(r'(?i)<kbd>', 'KBDSHIELDSTART', content) 

277 content = re.sub(r'(?i)</kbd>', 'KBDSHIELDEND', content) 

278 

279 # Ensure that we shield any custom patterns defined in the configuration, such as collapsibles or tabs,  

280 # before Pandoc sees them. 

281 content = content.replace('HASHSHIELD', '#') 

282 content = re.sub(r'<iframe.*?embed/([^"?\s]+).*?</iframe>', r'VIDEOTOKEN\1', content) 

283 

284 # Process any additional user-defined shielding patterns from the configuration file.  

285 # This allows for flexible extension of the shielding mechanism to accommodate various custom Markdown  

286 # constructs that may not be natively supported by Pandoc. 

287 patterns = self.conv_cfg.get("shielding_patterns", []) 

288 

289 # 9. We iterate through the list of shielding patterns defined in the configuration.  

290 # Each pattern is expected to have a "regex" to match, 

291 for p in patterns: 

292 # a "replacement" string that defines how to transform the matched content,  

293 # and optionally a "hook" that specifies 

294 regex = p.get("regex") 

295 replacement = p.get("replacement") 

296 if p.get("hook") == "protect_spaces": 

297 def protect_hook(match: re.Match) -> str: 

298 """ 

299 Function to protect spaces in titles by replacing them with a placeholder.  

300 This is necessary to prevent Pandoc from collapsing multiple spaces into one,  

301 which can cause issues with certain formatting or when the title is used as an ID. 

302 

303 Args: 

304 match: The regex match object containing the title and body to be protected. 

305 

306 Returns: 

307 str: The replacement string with spaces in the title protected by a placeholder. 

308 """ 

309 title = match.group(1).strip().replace(' ', 'PROTECTSPACE') 

310 body = match.group(2).strip() 

311 return replacement.replace(r"\1", title).replace(r"\2", body) 

312 content = re.sub(regex, protect_hook, content, flags=re.S) 

313 else: 

314 content = re.sub(regex, replacement, content, flags=re.S) 

315 

316 # 10. Finally, we shield JSON components like <JsonDisplay> to prevent any Pandoc interference,  

317 # using the same vault mechanism for safe restoration later. 

318 def shield_json_display(match: re.Match) -> str: 

319 """Shields <JsonDisplay> components to prevent Pandoc from mangling them.  

320 Uses a vault mechanism to store the original JSON content and replace it with a unique placeholder.  

321 

322 Args: 

323 match: The regex match object for a <JsonDisplay> component. 

324  

325 Returns: 

326 str: A unique placeholder string that will be used in the content to represent the original <JsonDisplay> component. 

327 """ 

328 placeholder = f"JSONP{len(self.protected_json)}PROTECT" 

329 self.protected_json.append(match.group(1)) 

330 return placeholder 

331 

332 # 11. We apply the shielding function to any <JsonDisplay> components found in the content,  

333 # ensuring that they are safely stored and replaced with unique placeholders before Pandoc processing. 

334 content = re.sub(r'(<JsonDisplay.*?\/>)', shield_json_display, content, flags=re.DOTALL) 

335 

336 # Protect HTML comments from being deleted by Pandoc 

337 # We concatenate the pattern so Markdown parsers don't hide it as a real comment! 

338 comment_pattern = r'<' + r'!--(.*?)-->' 

339 content = re.sub(comment_pattern, lambda m: f"HTMLCOMMENTSHIELD{m.group(1)}HTMLCOMMENTEND", content, flags=re.DOTALL) 

340 

341 # Shield custom IDs {#id}, but use a Negative Lookbehind (?<!\{)  

342 # so we DO NOT accidentally capture Handlebars block helpers like {{#if}} 

343 content = re.sub(r'(?<!\{)\{#(.*?)\}', r'IDSHIELDSTART\1IDSHIELDEND', content) 

344 

345 return content

348 def post_process_asciidoc(self, content: str) -> str: 

349 """ 

350 Restores shielded content, promotes admonitions, and finalizes headers in the AsciiDoc output. 

351 - Heading Anchors: Converts headers to Antora-compatible format with unique IDs. 

352 - The Cloak: Restores version headers that were vaulted to protect them from regex collisions. 

353 - HTML Restoration: Restores shielded HTML blocks and tags as passthroughs. 

354 - Cleanup & Metadata: Applies final regex cleanups and constructs the AsciiDoc header block with metadata. 

355  

356 Args: 

357 content (str): The raw AsciiDoc output from Pandoc. 

358  

359 Returns: 

360 str: The final, polished AsciiDoc content ready for Antora. 

361 """ 

362 # Guard clause for empty content to avoid unnecessary processing. 

363 self.used_ids = set() 

364 today = datetime.now().strftime("%Y-%m-%d") 

365 

366 # --- GLOBAL BUILD FLAGS --- 

367 # --- 1. ID STRATEGY TOGGLE --- 

368 # Set to True for SUSE-style [#id] shorthand 

369 # Set to False for long-form [id="id"] 

370 USE_SUSE_SHORTHAND = True 

371 # --- 2. DAPS COMPATIBILITY MODE --- 

372 DAPS_COMPATIBILITY_MODE = True 

373 

374 def smart_id_format(final_id: str) -> str: 

375 """ 

376 Always uses long-form [id="..."] if a dot exists to prevent build errors. 

377 Otherwise, respects the USE_SUSE_SHORTHAND flag. 

378 

379 Args: 

380 final_id (str): The ID to format. 

381 

382 Returns: 

383 str: The formatted ID block for AsciiDoc. 

384 """ 

385 if USE_SUSE_SHORTHAND and "." not in final_id: 

386 return f"[#{final_id}]" 

387 return f'[id="{final_id}"]' 

388 

389 # If the discovered title is something generic like "Untitled Document",  

390 # we won't slugify it to avoid generating an unhelpful ID. 

391 title_slug = self._slugify(self.discovered_title or "untitled") 

392 title_text = self._apply_global_attributes(self.discovered_title or "Untitled Document") 

393 

394 # --- 1. HEADING SLUGGING & NORMALIZATION --- 

395 def heading_anchor_logic(match: re.Match) -> str: 

396 """ 

397 Converts Markdown headers into Antora-compatible AsciiDoc headers with unique IDs. 

398 

399 Args: 

400 match: The regex match object for a header line, containing the header level and text. 

401 

402 Returns: 

403 str: The transformed header line with an Antora-compatible ID block. 

404 """ 

405 # We extract the header level characters (e.g., "==") and the raw title text from the matched header line. 

406 level_chars = match.group(1) 

407 raw_title = match.group(2).strip() 

408 

409 # We check if the raw title contains a custom ID shield (IDSHIELDSTART...IDSHIELDEND).  

410 # If it does, we extract the base ID from within the shield and use it as the final ID. 

411 custom_id_match = re.search(r'IDSHIELDSTART(.*?)IDSHIELDEND', raw_title) 

412 

413 # If a custom ID is found, we use it directly. If not, we generate a slug from the raw title text. 

414 if custom_id_match: 

415 # We extract the base ID from the custom ID shield and use it as the final ID for this header. 

416 base_id = custom_id_match.group(1) 

417 

418 # We also clean the display title by removing the custom ID shield from the raw title,  

419 # ensuring that the header text is clean and free of any ID artifacts. 

420 display_title = raw_title.replace(custom_id_match.group(0), "").strip() 

421 

422 # We must also ensure that the final ID is unique across the document.  

423 # If the base ID already exists in self.used_ids, we append a counter to it until we find a unique ID,  

424 # and then we register that final ID in self.used_ids to prevent future collisions. 

425 final_id = base_id 

426 

427 counter = 1 

428 

429 # If the final ID already exists, we append a counter until we find a unique one. 

430 while final_id in self.used_ids: 

431 final_id = f"{base_id}-{counter}" 

432 counter += 1 

433 self.used_ids.add(final_id) 

434 else: 

435 # If no custom ID is found, we generate a slug from the raw title text to use as the final ID for this header. 

436 final_id = self._slugify(raw_title) 

437 display_title = raw_title 

438 

439 # We apply global attribute replacements to the display title to ensure that any product names are  

440 # replaced with their corresponding Antora attributes, 

441 display_title = self._apply_global_attributes(display_title) 

442 

443 # Use the Smart Formatter to choose between [#id] and [id="id"] 

444 id_block = smart_id_format(final_id) 

445 

446 # Ensure only ONE return statement exists here 

447 return f'\n{id_block}\n{level_chars} {display_title}' 

448 

449 # Prepend a temporary newline to guarantee the absolute first line is caught 

450 content = "\n" + content 

451 

452 # Apply anchors to headers (Horizontal space [^\S\r\n]+ protects ==== admonition blocks) 

453 content = re.sub(r'\n(={2,6})[^\S\r\n]+([^\n]+)', heading_anchor_logic, content) 

454 

455 # --- 2. THE CLOAK --- 

456 if hasattr(self, 'heading_vault'): 

457 # We iterate through the heading vault and replace each unique key in the content with its original raw header text,  

458 # restoring the version headers that were protected from regex collisions during processing. 

459 for key, real_text in self.heading_vault.items(): 

460 # Replace the literal hyphen with an HTML entity 

461 # repair.py won't see it, but AsciiDoc will render it perfectly. 

462 cloaked_text = real_text.replace(" - ", " &#45; ") 

463 content = content.replace(key, cloaked_text) 

464 

465 # Restore JSON Components 

466 if hasattr(self, 'protected_json'): 

467 # We iterate through the list of protected JSON components and replace each unique placeholder in the content  

468 # with its original JSON content, ensuring that all <JsonDisplay> components are fully restored to their original form  

469 # after Pandoc processing. 

470 for i, original in enumerate(self.protected_json): 

471 content = content.replace(f"JSONP{i}PROTECT", original) 

472 

473 # --- 3. RESTORE HTML --- 

474 # Restore <style> as a block passthrough 

475 if hasattr(self, 'protected_html_blocks'): 

476 for i, original in enumerate(self.protected_html_blocks): 

477 # Wrap in ++++ so AsciiDoc renders the raw CSS block 

478 content = content.replace(f"HTMLBLOCKSHIELDA{i}A", f"++++\n{original}\n++++") 

479 

480 # Restore <div>, <ul>, <li> as inline passthroughs 

481 if hasattr(self, 'protected_html_tags'): 

482 for i, original in enumerate(self.protected_html_tags): 

483 # Wrap in +++ so AsciiDoc renders the raw tag 

484 content = content.replace(f"HTMLTAGSHIELDA{i}A", f"+++{original}+++") 

485 

486 # --- 4. CLEANUP & METADATA --- 

487 content = re.sub(r'IDSHIELDSTART.*?IDSHIELDEND', '', content) 

488 content = self._apply_global_attributes(content) 

489 

490 # UI MACROS (Keyboard) 

491 # 1. Convert KBD shields to AsciiDoc kbd:[] macros 

492 content = re.sub(r'KBDSHIELDSTART(.*?)KBDSHIELDEND', r'kbd:[\1]', content) 

493 

494 # 2. Merge adjacent kbd macros separated by a plus OR Pandoc's {plus} 

495 content = re.sub(r'kbd:\[([^\]]+)\]\s*(?:\+|\{plus\})\s*kbd:\[([^\]]+)\]', r'kbd:[\1+\2]', content) 

496 content = re.sub(r'kbd:\[([^\]]+)\]\s*(?:\+|\{plus\})\s*kbd:\[([^\]]+)\]', r'kbd:[\1+\2]', content) 

497 

498 # Header Block: Uses the same smart ID logic for the Title 

499 main_heading = "==" if DAPS_COMPATIBILITY_MODE else "=" 

500 header_lines = [ 

501 smart_id_format(title_slug), 

502 f"{main_heading} {title_text}", 

503 ":idprefix:", 

504 ":idseparator: -" 

505 ] 

506 

507 # Inject doctype (defaults to 'article' if not set via CLI) 

508 doctype = getattr(self, 'doctype', 'article') 

509 if doctype != 'article': 

510 header_lines.append(f":doctype: {doctype}") 

511 

512 # Inject the experimental flag if UI macros like kbd:[] are detected in the content 

513 if "kbd:[" in content: 

514 header_lines.append(":experimental:") 

515 

516 # Apply branding attributes to metadata values as well 

517 for key, value in self.metadata.items(): 

518 if key.lower() == "title": 

519 continue 

520 

521 # Check the Master Toggle first 

522 if not self.REPLACE_WITH_ATTRIBUTES: 

523 header_lines.append(f":{key}: {value}") 

524 continue 

525 

526 # CASE A: The value is a simple String (like Description) 

527 if isinstance(value, str): 

528 processed_value = self._apply_global_attributes(value) 

529 header_lines.append(f":{key}: {processed_value}") 

530 

531 # CASE B: The value is a List (like Keywords) 

532 elif isinstance(value, list): 

533 # We process every string inside the list 

534 processed_list = [ 

535 self._apply_global_attributes(item) if isinstance(item, str) else item 

536 for item in value 

537 ]

631 """

662 else: 

663 # For any other restoration patterns that don't have a specific hook,  

664 # we apply a standard regex substitution using the provided pattern and replacement string,

669 if mapping:

769 def antora_xref_logic(m: re.Match) -> str: 

770 """ 

771 Converts Markdown links into Antora-compatible xrefs, handling various path formats and anchors. 

772 

773 Args: 

774 m: The regex match object for a Markdown link, containing the raw path and optional anchor 

775 Returns: 

776 str: The transformed link in Antora xref format, with the path adjusted to remove file extensions and anchors formatted correctly. 

777 """ 

778 

779 # We extract the raw path and optional anchor from the matched Markdown link.  

780 # The raw path may include file extensions like .md or .adoc, which we will remove to convert it to an Antora xref format. The anchor, if present, will be formatted to ensure it is compatible with Antora's linking system. 

781 raw_path = m.group(1) or "" 

782 anchor = m.group(2) or "" 

783 

784 # We process the raw path to remove any .md or .adoc extensions, as Antora xrefs should not include file extensions. 

785 path = raw_path.replace(".md", "").replace(".adoc", "").replace("./", "").strip("/") 

786 

787 # We also check if the path contains "inputs/", which is a common pattern for Antora includes.  

788 # If it does, we remove the "inputs/" prefix to convert it to a proper xref path. 

789 if "inputs/" in path: 

790 path = path.split("inputs/")[-1] 

791 if path: 

792 path = f"{path}.adoc" 

793 cl_anchor = "" 

794 if anchor: 

795 cl_anchor = "#" + anchor.replace("#", "").lower() 

796 return f"xref:{path}{cl_anchor}" 

797 

798 # We apply the antora_xref_logic function to any Markdown links that match the pattern for link: or xref:,  

799 # converting them into Antora-compatible xrefs by adjusting the path to remove file extensions and  

800 # formatting any anchors correctly, ensuring that all internal links are properly transformed for use in Antora. 

801 content = re.sub(r'(?:link:|xref:)(?:\+\+)?([^\[\s#\+]+)?(#[^\[\s\+]+)?(?:\+\+)?', antora_xref_logic, content) 

802 content = re.sub(r'\[source,mermaid\]\n----(.*?)----', r'[mermaid]\n....\1....', content, flags=re.DOTALL) 

803 content = content.replace("SHIELDADMONSTARTtabs", "[tabs]\n====") 

804 content = content.replace("SHIELDADMONEND", "====") 

805 content = re.sub(r'^@tab\s+(.*)$', r'\1::', content, flags=re.M) 

806 

807 # --- 9. DAPS COMPATIBILITY MODE --- 

808 # Set to True if building with DAPS (DocBook XML requires specific escaping and header nesting). 

809 # Set to False if building with Antora or standard Asciidoctor. 

810 DAPS_COMPATIBILITY_MODE = True 

811 

812 if DAPS_COMPATIBILITY_MODE: 

813 # A. HEADING DEMOTION 

814 # DAPS/DocBook crashes if included modular files start with a Level 0 (=) Book Part header. 

815 # This shifts all headings down by one level (= becomes ==, == becomes ===) 

816 content = re.sub(r'^(={1,5})[ \t]+', r'\1= ', content, flags=re.MULTILINE) 

817 

818 # B. ESCAPE HANDLEBARS 

819 # Escape Handlebars templates ({{ }}) so DAPS doesn't mistake them for unresolvable attributes. 

820 content = content.replace("{{{", "\\{\\{\\{").replace("{{", "\\{\\{") 

821 content = content.replace("++\\{\\{\\{++", "\\{\\{\\{").replace("++\\{\\{++", "\\{\\{") 

822 

823 # --- 10. FINAL ARTIFACT CLEANUP --- 

824 # These are general transpiler artifacts that need to be cleaned up regardless of the build system. 

825 

826 # Clean up custom admonition artifacts (.BODY and ENDADMON) 

827 content = re.sub( 

828 r'\s*\.BODY(.*?)(?:\.?ENDADMON)', 

829 lambda m: f"\n====\n{m.group(1).strip()}\n====", 

830 content, 

831 flags=re.S 

832 ) 

833 

834 # Clean up leaked Docusaurus Admonition Shields 

835 content = re.sub(r'SHIELDADMONSTART([a-zA-Z]+)TITLE\s*', lambda m: f"[{m.group(1).upper()}]\n", content) 

836 

837 # Restore HTML comments as AsciiDoc multi-line comment blocks (////) 

838 content = re.sub( 

839 r'HTMLCOMMENTSHIELD(.*?)HTMLCOMMENTEND', 

840 lambda m: f"////\n{m.group(1).strip()}\n////\n", 

841 content, 

842 flags=re.DOTALL 

843 ) 

844 

845 return header_block + content.strip()

848 def convert_file(self, input_path: Path, output_path: Path) -> None: 

849 """ 

850 Orchestrates the conversion of a single Markdown file to AsciiDoc. 

851  

852 Args: 

853 input_path (Path): Source Markdown file. 

854 output_path (Path): Destination for the raw AsciiDoc. 

855  

856 Returns: 

857 None: Writes the converted content to output_path. 

858 """ 

859 # We reset the metadata and discovered title for each file conversion to ensure that we start with a clean slate  

860 # for each document, preventing any carryover of metadata or titles from previous conversions that could lead to  

861 # incorrect or duplicated information in the final AsciiDoc output. 

862 self.metadata = {} 

863 self.discovered_title = None 

864 

865 # 1. Read the raw Markdown content from the input file using UTF-8 encoding to ensure proper handling of  

866 # special characters and international text. 

867 raw_md = input_path.read_text(encoding='utf-8') 

868 ready_md = self.pre_process_markdown(raw_md) 

869 

870 # 2. We write to a temporary file so Pandoc sees the 'shielded' version 

871 temp_md = input_path.with_suffix('.tmp.md') 

872 temp_md.write_text(ready_md, encoding='utf-8') 

873 

874 # 3. We use subprocess to call the Pandoc CLI for conversion, specifying the input format  

875 # as "markdown-smart" and the output format as "asciidoc". 

876 try: 

877 # Execute Pandoc CLI 

878 subprocess.run( 

879 [ 

880 "pandoc", 

881 "-f", "markdown-smart", 

882 "-t", "asciidoc", 

883 "--shift-heading-level-by=-1", 

884 "--wrap=none", 

885 "-o", str(output_path), 

886 str(temp_md) 

887 ], 

888 check=True, 

889 capture_output=True 

890 ) 

891 

892 # Post-process the Pandoc result to restore shields and finalize headers 

893 final_adoc = self.post_process_asciidoc(output_path.read_text(encoding='utf-8')) 

894 output_path.write_text(final_adoc, encoding='utf-8') 

895 finally: 

896 # Tidy up transient files 

897 if temp_md.exists(): 

898 temp_md.unlink()
