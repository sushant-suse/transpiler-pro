 70    def _apply_global_attributes(self, text: str) -> str:
 71        """
 72        Replaces raw product names with Antora attributes.
 73        Uses negative lookbehind/lookahead to protect URLs and file paths.
 74
 75        Args:
 76            text (str): The input string to process.
 77
 78        Returns:
 79            str: The text with product names replaced by attributes.
 80        """
 81        # Guard clause: If text is empty or if attribute replacement is disabled, return the original text.
 82        if not text or not self.REPLACE_WITH_ATTRIBUTES:
 83            return text
 84        
 85        # We iterate through the attribute map and apply replacements. 
 86        # The regex ensures we only replace standalone occurrences of the product names, 
 87        # not when they are part of URLs or file paths.        
 88        for raw_name, attr in self.attribute_map.items():
 89            # SMART REGEX:
 90            # (?<![/:]) -> Negative Lookbehind: Don't match if preceded by / or : (Shields paths/URLs)
 91            # \b...\b   -> Word Boundary: Match the whole word only
 92            # (?![/.])  -> Negative Lookahead: Don't match if followed by / or . (Shields extensions/paths)
 93            pattern = rf"(?<![/:])\b{re.escape(raw_name)}\b(?![/.])"
 94            
 95            text = re.sub(pattern, attr, text)
 96
 97        return text

100    def _slugify(self, text: str) -> str:
101        """
102        Converts a heading title into a SEO-friendly, unique ID.
103
104        Args:
105            text (str): The heading text to slugify.
106        Returns:
107            str: A slugified version of the heading, suitable for use as an ID.
108        """
109        # Guard clause for empty text to prevent generating empty IDs
110        lookup_key = text.strip()
111        
112        if "VLT" in lookup_key and "HVAULT" in lookup_key:
113            if hasattr(self, 'heading_vault') and lookup_key in self.heading_vault:
114                text = self.heading_vault[lookup_key]
115
116        # 1. Strip raw HTML tags (if any survived)
117        clean_text = re.sub(r'<[^>]+>', '', text)
118        
119        # 2. Strip AsciiDoc roles (e.g., [.class-name]#Text#)
120        clean_text = re.sub(r'\[\.[^\]]+\]#.*?#', '', clean_text)
121        
122        # 3. Normalize to lowercase and replace spaces with hyphens
123        slug = clean_text.lower()
124        
125        # 4. Remove any remaining curly brace IDs and replace dots with hyphens
126        slug = re.sub(r'\{#.*?\}', '', slug) 
127        slug = slug.replace('.', '-')
128        slug = re.sub(r'[^a-z0-9-]', '-', slug)
129        slug = re.sub(r'-+', '-', slug).strip('-')
130        
131        # 5. Ensure uniqueness by appending a counter if needed
132        base_slug = slug or "section"
133        final_slug = base_slug
134        
135        counter = 1
136
137        # 6. If the slug already exists, append a counter until we find a unique one
138        while final_slug in self.used_ids:
139            final_slug = f"{base_slug}-{counter}"
140            counter += 1
141        
142        # 7. Register the final slug to prevent future collisions
143        self.used_ids.add(final_slug)
144
145        return final_slug

148    def pre_process_markdown(self, content: str) -> str:
149        """
150        Prepares Markdown for Pandoc by shielding modern syntax and extracting metadata.
151        Includes a 'Vault' mechanism to protect version headers from line-splitting.
152
153        Args:
154            content (str): Raw Markdown string.
155
156        Returns:
157            str: "Shielded" Markdown ready for Pandoc.
158        """
159        self.metadata = {}
160        self.discovered_title = None
161        self.used_ids = set()
162        self.protected_json = []
163        self.heading_vault = {}
164
165        def vault_header(m: re.Match) -> str:
166            """
167            Protects version headers like "## 2.2.0 - 2026-03-23" from being split by the list item regex.
168
169            Args:
170                m: The regex match object for a version header.
171            """
172            # We store the raw header text in a vault and replace it with a unique key.
173            level = m.group(1)
174
175            # We want to preserve the exact text of the header, including the version and date, so we store it in the vault.
176            raw_text = m.group(2).strip()
177
178            # We generate a unique key for this header and store the raw text in the vault under that key.
179            key = f"VLT{len(self.heading_vault)}HVAULT"
180
181            # We store the raw header text in the vault using the unique key.
182            self.heading_vault[key] = raw_text
183
184            return f"{level} {key}"
185
186        # Secure the version headers before anything else touches them
187        # We look for headers that start with ## (or more) followed by a version pattern and a date, and we vault them.
188        pattern = r'^(#+)\s+(\d+\.[\d\.]+\d+\s+-\s+\d{4}-\d{2}-\d{2}).*$'
189        # We replace the matched header with a vault key, and store the original header text 
190        # in the heading_vault for later restoration.
191        content = re.sub(pattern, vault_header, content, flags=re.M)
192
193        # --- EXISTING SHIELDING LOGIC ---
194        # 1. Code Blocks: Shield fenced code blocks to prevent Pandoc from mangling them.
195        content = re.sub(r'(`{3}.*?`{3})', lambda m: m.group(1).replace('#', 'HASHSHIELD'), content, flags=re.DOTALL)
196        # 2. Inline Code: Shield inline code to prevent accidental ID processing.
197        content = re.sub(r'(`.*?`)', lambda m: m.group(1).replace('#', 'HASHSHIELD'), content)
198
199        # 3. Iframes: Temporarily replace iframes with a token to prevent Pandoc from stripping them.
200        frontmatter_match = re.match(r'^---\s*\n(.*?)\n---\s*\n', content, re.DOTALL)
201
202        # 4. Custom Shielding Patterns: Apply any user-defined regex patterns for shielding.
203        if frontmatter_match:
204            try:
205                import yaml
206
207                # We attempt to parse the frontmatter as YAML to extract metadata. 
208                # If successful, we store it in self.metadata.
209                # If the frontmatter is not valid YAML, we catch the exception and simply leave 
210                # self.metadata as an empty dictionary, allowing the conversion to proceed without metadata.
211                yaml_data = yaml.safe_load(frontmatter_match.group(1))
212
213                # We check if the parsed YAML data is a dictionary before assigning it to self.metadata.
214                if isinstance(yaml_data, dict):
215                    self.metadata = yaml_data
216
217                # If a title is found in the metadata, we can remove it from the content to prevent duplication.
218                content = content[frontmatter_match.end():]
219            except Exception:
220                # If YAML parsing fails, we simply ignore the frontmatter and proceed with an empty metadata dictionary.
221                self.metadata = {}
222
223        # 5. Title Discovery: If no title is in metadata, infer from the first H1 and remove it from content.
224        self.discovered_title = self.metadata.get('title')
225
226        # We look for the first H1 header in the content to use as the title if it wasn't provided in the metadata.
227        if not self.discovered_title:
228            # We search for a line that starts with a single '#' followed by the title text.
229            h1_match = re.search(r'^#\s+(.*)$', content, re.M)
230
231            # If we find an H1 header, we set it as the discovered title and remove that line 
232            # from the content to prevent duplication in the final AsciiDoc.
233            if h1_match:
234                self.discovered_title = h1_match.group(1).strip()
235                # We remove the first occurrence of this H1 header from the content.
236                content = content.replace(h1_match.group(0), "", 1)
237
238        # --- 6. HTML SHIELDING ---
239        self.protected_html_blocks = []
240        def shield_html_block(match):
241            placeholder = f"HTMLBLOCKSHIELDA{len(self.protected_html_blocks)}A"
242            self.protected_html_blocks.append(match.group(1))
243            return placeholder
244        
245        # Shield entire <style> or <script> blocks so Pandoc ignores their contents
246        content = re.sub(r'(<style.*?</style>)', shield_html_block, content, flags=re.DOTALL|re.IGNORECASE)
247
248        # Additionally, we will shield structural HTML tags like <div>, <ul>, and <li> to prevent Pandoc from stripping them.
249        self.protected_html_tags = []
250
251        # 7. Shielding Structural HTML Tags: We define a function to shield structural HTML tags like <div>, <ul>, and <li>.
252        def shield_layout_tag(match: re.Match) -> str:
253            """
254            Shields structural HTML tags like <div>, <ul>, and <li> to prevent Pandoc from stripping them.
255            Uses a vault mechanism similar to the version headers to store the original tag content and 
256            replace it with a unique placeholder. This allows us to restore the exact original HTML tags after Pandoc processing,
257            without risking any regex collisions or unintended replacements.
258
259            The placeholder format is designed to be unique and easily identifiable for restoration in the post-processing phase 
260            without interfering with any other content. By using a vault, we can ensure that even if the same HTML tag appears multiple times, 
261            each instance will be stored and restored correctly without any risk of collision or misreplacement.   
262
263            Args:
264                match: The regex match object for an HTML tag to be shielded.
265
266            Returns:
267                str: A unique placeholder string that will be used in the content to represent the original HTML tag. 
268            """
269            placeholder = f"HTMLTAGSHIELDA{len(self.protected_html_tags)}A"
270            self.protected_html_tags.append(match.group(1))
271            return placeholder
272
273        # 8a. Shield structural tags (div, ul, li) but let Pandoc handle <span> natively
274        content = re.sub(r'(</?(?:div|ul|li)[^>]*>)', shield_layout_tag, content, flags=re.IGNORECASE)
275
276        # 8b. Shield <kbd> tags safely without applying HTML passthroughs (+++)
277        content = re.sub(r'(?i)<kbd>', 'KBDSHIELDSTART', content)
278        content = re.sub(r'(?i)</kbd>', 'KBDSHIELDEND', content)
279
280        # Ensure that we shield any custom patterns defined in the configuration, such as collapsibles or tabs, 
281        # before Pandoc sees them.
282        content = content.replace('HASHSHIELD', '#')
283        content = re.sub(r'<iframe.*?embed/([^"?\s]+).*?</iframe>', r'VIDEOTOKEN\1', content)
284
285        # Process any additional user-defined shielding patterns from the configuration file. 
286        # This allows for flexible extension of the shielding mechanism to accommodate various custom Markdown 
287        # constructs that may not be natively supported by Pandoc.
288        patterns = self.conv_cfg.get("shielding_patterns", [])
289
290        # 9. We iterate through the list of shielding patterns defined in the configuration. 
291        # Each pattern is expected to have a "regex" to match,
292        for p in patterns:
293            # a "replacement" string that defines how to transform the matched content, 
294            # and optionally a "hook" that specifies
295            regex = p.get("regex")
296            replacement = p.get("replacement")
297            if p.get("hook") == "protect_spaces":
298                def protect_hook(match: re.Match) -> str:
299                    """
300                    Function to protect spaces in titles by replacing them with a placeholder. 
301                    This is necessary to prevent Pandoc from collapsing multiple spaces into one, 
302                    which can cause issues with certain formatting or when the title is used as an ID.
303
304                    Args:
305                        match: The regex match object containing the title and body to be protected.
306
307                    Returns:
308                        str: The replacement string with spaces in the title protected by a placeholder.
309                    """
310                    title = match.group(1).strip().replace(' ', 'PROTECTSPACE')
311                    body = match.group(2).strip()
312                    return replacement.replace(r"\1", title).replace(r"\2", body)
313                content = re.sub(regex, protect_hook, content, flags=re.S)
314            else:
315                content = re.sub(regex, replacement, content, flags=re.S)
316        
317        # 10. Finally, we shield JSON components like <JsonDisplay> to prevent any Pandoc interference, 
318        # using the same vault mechanism for safe restoration later.
319        def shield_json_display(match: re.Match) -> str:
320            """Shields <JsonDisplay> components to prevent Pandoc from mangling them. 
321            Uses a vault mechanism to store the original JSON content and replace it with a unique placeholder. 
322
323            Args:
324                match: The regex match object for a <JsonDisplay> component.
325            
326            Returns:
327                str: A unique placeholder string that will be used in the content to represent the original <JsonDisplay> component.
328            """
329            placeholder = f"JSONP{len(self.protected_json)}PROTECT"
330            self.protected_json.append(match.group(1))
331            return placeholder
332
333        # 11. We apply the shielding function to any <JsonDisplay> components found in the content, 
334        # ensuring that they are safely stored and replaced with unique placeholders before Pandoc processing.
335        content = re.sub(r'(<JsonDisplay.*?\/>)', shield_json_display, content, flags=re.DOTALL)
336        
337        # Protect HTML comments from being deleted by Pandoc
338        # We concatenate the pattern so Markdown parsers don't hide it as a real comment!
339        comment_pattern = r'<' + r'!--(.*?)-->'
340        content = re.sub(comment_pattern, lambda m: f"HTMLCOMMENTSHIELD{m.group(1)}HTMLCOMMENTEND", content, flags=re.DOTALL)
341
342        # Shield custom IDs {#id}, but use a Negative Lookbehind (?<!\{) 
343        # so we DO NOT accidentally capture Handlebars block helpers like {{#if}}
344        content = re.sub(r'(?<!\{)\{#(.*?)\}', r'IDSHIELDSTART\1IDSHIELDEND', content)
345
346        return content

349    def post_process_asciidoc(self, content: str) -> str:
350        """
351        Restores shielded content, promotes admonitions, and finalizes headers in the AsciiDoc output.
352            - Heading Anchors: Converts headers to Antora-compatible format with unique IDs.
353            - The Cloak: Restores version headers that were vaulted to protect them from regex collisions.
354            - HTML Restoration: Restores shielded HTML blocks and tags as passthroughs.
355            - Cleanup & Metadata: Applies final regex cleanups and constructs the AsciiDoc header block with metadata.
356        
357        Args:
358            content (str): The raw AsciiDoc output from Pandoc.
359        
360        Returns:
361            str: The final, polished AsciiDoc content ready for Antora.
362        """
363        # Guard clause for empty content to avoid unnecessary processing.
364        self.used_ids = set() 
365        today = datetime.now().strftime("%Y-%m-%d")
366
367        # --- GLOBAL BUILD FLAGS ---
368        # --- 1. ID STRATEGY TOGGLE ---
369        # Set to True for SUSE-style [#id] shorthand
370        # Set to False for long-form [id="id"]
371        USE_SUSE_SHORTHAND = True
372        # --- 2. DAPS COMPATIBILITY MODE ---
373        DAPS_COMPATIBILITY_MODE = True
374
375        def smart_id_format(final_id: str) -> str:
376            """
377            Always uses long-form [id="..."] if a dot exists to prevent build errors.
378            Otherwise, respects the USE_SUSE_SHORTHAND flag.
379
380            Args:
381                final_id (str): The ID to format.
382
383            Returns:
384                str: The formatted ID block for AsciiDoc.
385            """
386            if USE_SUSE_SHORTHAND and "." not in final_id:
387                return f"[#{final_id}]"
388            return f'[id="{final_id}"]'
389        
390        # If the discovered title is something generic like "Untitled Document", 
391        # we won't slugify it to avoid generating an unhelpful ID.
392        title_slug = self._slugify(self.discovered_title or "untitled")
393        title_text = self._apply_global_attributes(self.discovered_title or "Untitled Document")
394
395        # --- 1. HEADING SLUGGING & NORMALIZATION ---
396        def heading_anchor_logic(match: re.Match) -> str:
397            """
398            Converts Markdown headers into Antora-compatible AsciiDoc headers with unique IDs.
399
400            Args:
401                match: The regex match object for a header line, containing the header level and text.
402
403            Returns:
404                str: The transformed header line with an Antora-compatible ID block.
405            """
406            # We extract the header level characters (e.g., "==") and the raw title text from the matched header line.
407            level_chars = match.group(1)
408            raw_title = match.group(2).strip()
409            
410            # We check if the raw title contains a custom ID shield (IDSHIELDSTART...IDSHIELDEND). 
411            # If it does, we extract the base ID from within the shield and use it as the final ID.
412            custom_id_match = re.search(r'IDSHIELDSTART(.*?)IDSHIELDEND', raw_title)
413
414            # If a custom ID is found, we use it directly. If not, we generate a slug from the raw title text.
415            if custom_id_match:
416                # We extract the base ID from the custom ID shield and use it as the final ID for this header.
417                base_id = custom_id_match.group(1)
418
419                # We also clean the display title by removing the custom ID shield from the raw title, 
420                # ensuring that the header text is clean and free of any ID artifacts.
421                display_title = raw_title.replace(custom_id_match.group(0), "").strip()
422
423                # We must also ensure that the final ID is unique across the document. 
424                # If the base ID already exists in self.used_ids, we append a counter to it until we find a unique ID, 
425                # and then we register that final ID in self.used_ids to prevent future collisions.
426                final_id = base_id
427                
428                counter = 1
429
430                # If the final ID already exists, we append a counter until we find a unique one.
431                while final_id in self.used_ids:
432                    final_id = f"{base_id}-{counter}"
433                    counter += 1
434                self.used_ids.add(final_id)
435            else:
436                # If no custom ID is found, we generate a slug from the raw title text to use as the final ID for this header.
437                final_id = self._slugify(raw_title)
438                display_title = raw_title
439
440            # We apply global attribute replacements to the display title to ensure that any product names are 
441            # replaced with their corresponding Antora attributes,
442            display_title = self._apply_global_attributes(display_title)
443            
444            # Use the Smart Formatter to choose between [#id] and [id="id"]
445            id_block = smart_id_format(final_id)
446            
447            # Ensure only ONE return statement exists here
448            return f'\n{id_block}\n{level_chars} {display_title}'
449
450        # Prepend a temporary newline to guarantee the absolute first line is caught
451        content = "\n" + content
452
453        # Apply anchors to headers (Horizontal space [^\S\r\n]+ protects ==== admonition blocks)
454        content = re.sub(r'\n(={2,6})[^\S\r\n]+([^\n]+)', heading_anchor_logic, content)
455
456        # --- 2. THE CLOAK ---
457        if hasattr(self, 'heading_vault'):
458            # We iterate through the heading vault and replace each unique key in the content with its original raw header text, 
459            # restoring the version headers that were protected from regex collisions during processing.
460            for key, real_text in self.heading_vault.items():
461                # Replace the literal hyphen with an HTML entity
462                # repair.py won't see it, but AsciiDoc will render it perfectly.
463                cloaked_text = real_text.replace(" - ", " &#45; ")
464                content = content.replace(key, cloaked_text)
465        
466        # Restore JSON Components
467        if hasattr(self, 'protected_json'):
468            # We iterate through the list of protected JSON components and replace each unique placeholder in the content 
469            # with its original JSON content, ensuring that all <JsonDisplay> components are fully restored to their original form 
470            # after Pandoc processing.
471            for i, original in enumerate(self.protected_json):
472                content = content.replace(f"JSONP{i}PROTECT", original)
473
474        # --- 3. RESTORE HTML ---
475        # Restore <style> as a block passthrough
476        if hasattr(self, 'protected_html_blocks'):
477            for i, original in enumerate(self.protected_html_blocks):
478                # Wrap in ++++ so AsciiDoc renders the raw CSS block
479                content = content.replace(f"HTMLBLOCKSHIELDA{i}A", f"++++\n{original}\n++++")
480                
481        # Restore <div>, <ul>, <li> as inline passthroughs
482        if hasattr(self, 'protected_html_tags'):
483            for i, original in enumerate(self.protected_html_tags):
484                # Wrap in +++ so AsciiDoc renders the raw tag
485                content = content.replace(f"HTMLTAGSHIELDA{i}A", f"+++{original}+++")
486
487        # --- 4. CLEANUP & METADATA ---
488        content = re.sub(r'IDSHIELDSTART.*?IDSHIELDEND', '', content)
489        content = self._apply_global_attributes(content)
490
491        # UI MACROS (Keyboard)
492        # 1. Convert KBD shields to AsciiDoc kbd:[] macros
493        content = re.sub(r'KBDSHIELDSTART(.*?)KBDSHIELDEND', r'kbd:[\1]', content)
494        
495        # 2. Merge adjacent kbd macros separated by a plus OR Pandoc's {plus}
496        content = re.sub(r'kbd:\[([^\]]+)\]\s*(?:\+|\{plus\})\s*kbd:\[([^\]]+)\]', r'kbd:[\1+\2]', content)
497        content = re.sub(r'kbd:\[([^\]]+)\]\s*(?:\+|\{plus\})\s*kbd:\[([^\]]+)\]', r'kbd:[\1+\2]', content)
498
499        # Header Block: Uses the same smart ID logic for the Title
500        main_heading = "==" if DAPS_COMPATIBILITY_MODE else "="
501        header_lines = [
502            smart_id_format(title_slug),
503            f"{main_heading} {title_text}",
504            ":idprefix:",
505            ":idseparator: -"
506        ]
507

 86    def _get_progressive_verb(self, verb_token) -> str:
 87        """
 88        Logic to convert a verb to its '-ing' form.
 89        
 90        Prioritizes the 'special_verbs' table in pyproject.toml to handle 
 91        irregular conjugations (e.g., 'stop' -> 'stopping') before falling 
 92        back to standard English suffix rules.
 93        """
 94        lemma = verb_token.lemma_.lower()
 95        grammar_cfg = self.config.get("grammar", {})
 96        special = grammar_cfg.get("special_verbs", {})
 97        
 98        if lemma in special:
 99            return special[lemma]
100
101        # Standard -ing rules
102        if lemma.endswith("e") and not lemma.endswith("ee"):
103            return lemma[:-1] + "ing"
104        # CVC rule: Double the consonant (e.g., run -> running)
105        if len(lemma) > 2 and lemma[-1] not in "aeiou" and lemma[-2] in "aeiou" and lemma[-3] not in "aeiou":
106            return lemma + lemma[-1] + "ing"
107        return lemma + "ing"

109    def _fix_tense(self, line: str) -> str:
110        """
111        Standard Tense Shifter: "We will test" -> "We are testing".
112        Note: This is an legacy/alternative shifter; primary tense shifting 
113        is now handled by the more advanced LinguisticEngine in repair.py.
114        """
115        if not self.nlp: 
116            return line
117        doc = self.nlp(line)
118        working_line = line
119        for token in doc:
120            if token.text.lower() == "will":
121                main_verb = token.head
122                if main_verb.pos_ == "VERB":
123                    # Determine plurality for correct aux verb (is vs are)
124                    subjects = [w for w in main_verb.lefts if "subj" in w.dep_]
125                    is_plural = any("Number=Plur" in str(s.morph) or s.text.lower() in ["we", "they", "you"] for s in subjects)
126                    aux = "are" if is_plural else "is"
127                    prog = self._get_progressive_verb(main_verb)
128                    working_line = re.sub(rf"\b{token.text}\s+{main_verb.text}\b", f"{aux} {prog}", working_line, flags=re.IGNORECASE)
129        return working_line

171            for issue in line_map[line_num]:
172                msg = issue.get("Message", "")
173                check_id = issue.get("Check", "")
174                suggestion = issue.get("Suggestion", "")
175

177                for wrong, correct in session_branding.items():

185                        working_line = re.sub(rf"\b{re.escape(target)}\b\s?", "", working_line, flags=re.IGNORECASE)
186

191                        wrong_term = m[1] if len(m) >= 2 else (m[0] if m else "")

194                            if "will" in wrong_term.lower() or "will" in msg.lower():
195                                continue

197
198                # 4. Auto-Learning: Capture spelling fixes into the Knowledge Base
199                elif "Spelling" in check_id:
200                    if suggestion and suggestion.lower() not in ["spelling", "spellings", "learned"]:

203                        if word_to_fix:

221
222            # Update line in content if modifications were made
223            if working_line != original_line:
224                content[idx] = working_line
225                total_fixes += 1
226
227        # Write corrected content back and update the JSON brain
228        file_path.write_text("\n".join(content), encoding="utf-8")
229        self._save_kb()
